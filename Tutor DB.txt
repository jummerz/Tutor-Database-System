CREATE TABLE Student (
Student_ID NUMBER (3), 
First_Name VARCHAR2(30),
Last_Name VARCHAR2(30),
Phone_Number VARCHAR2 (15), 
Email VARCHAR2(30),
CONSTRAINT PK_Student PRIMARY KEY(Student_ID)
);

CREATE TABLE Tutor (
	Tutor_ID NUMBER(3),
	First_Name VARCHAR2(25),
	Last_Name VARCHAR2(25),
	Phone_Number VARCHAR2(15),
	Session_Cost NUMERIC(5,2),
	Email VARCHAR2(50),
	Total_Completed_Sessions NUMBER(30),
	CONSTRAINT PK_Tutor PRIMARY KEY(Tutor_ID)
);

CREATE TABLE Appointment (
	Appointment_ID NUMBER(3),
	Student_ID NUMBER (3), 
	Tutor_ID NUMBER(3),
	Appt_Date DATE,
	Time VARCHAR2(30),
	Room_Number VARCHAR(30),
	Status VARCHAR(30),
	Payment_amount NUMERIC (5,2),
	Rating NUMBER(2),
	Student_Comment VARCHAR(200),
	Review_date DATE,
CONSTRAINT Appointment_PK PRIMARY KEY (Appointment_ID),
CONSTRAINT Appointment_FK1 FOREIGN KEY (Student_ID) REFERENCES Student(Student_ID),
CONSTRAINT Appointment_FK2 FOREIGN KEY (Tutor_ID) REFERENCES
Tutor(Tutor_ID)
);

CREATE TABLE Availability (
	Availability_ID NUMBER(3),
	Tutor_ID NUMBER(3),
	Day_of_week VARCHAR2(10),
	Time VARCHAR2(10),
	Status VARCHAR2(20),
	CONSTRAINT Availability_PK PRIMARY KEY (Availability_ID, Tutor_ID),
	CONSTRAINT Availability_FK FOREIGN KEY (Tutor_ID) REFERENCES Tutor(Tutor_ID)
);
